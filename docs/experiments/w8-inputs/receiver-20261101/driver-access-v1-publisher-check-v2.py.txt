import hashlib,json,os,pathlib,resource,stat,sys
resource.setrlimit(resource.RLIMIT_CPU,(1,1));p=pathlib.Path(sys.argv[1]);want=json.loads(sys.argv[2]);out={}
if set(x.name for x in p.iterdir())!=set(want):raise ValueError('directory membership')
for name,digest in want.items():
 q=p/name; s=q.lstat()
 if not stat.S_ISREG(s.st_mode) or stat.S_IMODE(s.st_mode)!=0o444:raise ValueError('mode or type')
 fd=os.open(q,os.O_RDONLY|os.O_NOFOLLOW)
 try:
  before=os.fstat(fd); h=hashlib.sha256()
  while True:
   b=os.read(fd,65536)
   if not b:break
   h.update(b)
  after=os.fstat(fd); end=q.lstat()
 finally:os.close(fd)
 fields=lambda x:(x.st_dev,x.st_ino,x.st_size,x.st_mtime_ns,x.st_ctime_ns,x.st_mode)
 if fields(s)!=fields(before) or fields(before)!=fields(after) or fields(after)!=fields(end) or h.hexdigest()!=digest:raise ValueError('identity or unstable file')
 out[name]={'sha256':digest,'size':end.st_size,'device':end.st_dev,'inode':end.st_ino,'mtime_ns':end.st_mtime_ns,'ctime_ns':end.st_ctime_ns}
print(json.dumps({'passed':True,'files':out},sort_keys=True))
