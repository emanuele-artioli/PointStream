import hashlib,json,os,resource,signal,subprocess,time
from pathlib import Path
HERE=Path(__file__).resolve().parent
OUT=Path('/private/tmp/w8-cost-source-repair-cpu-v1-20261101')
EXPECTED='848317d21bc9218f784fd8c5c574ddbd96cf905c5308d2968691cdc4ca5b9219'
resource.setrlimit(resource.RLIMIT_CPU,(2,2))
command=(HERE/'cost-source-repair-cpu-v1-command.txt').read_bytes()
if hashlib.sha256(command).hexdigest()!=EXPECTED or len(command)>=120000 or len(command)+sum(len(k)+len(v)+2 for k,v in os.environ.items())>=200000:raise ValueError('command hash or size')
OUT.mkdir(mode=0o700,exist_ok=False)
def alarm(*_):raise TimeoutError()
def limit():resource.setrlimit(resource.RLIMIT_CPU,(1,1))
signal.signal(signal.SIGALRM,alarm);signal.alarm(95);started=time.monotonic()
with (OUT/'stdout').open('xb') as stdout,(OUT/'stderr').open('xb') as stderr:
 proc=subprocess.Popen(['/usr/bin/ssh','-o','BatchMode=yes','-o','ConnectTimeout=5','gpu6',command.decode()],stdin=subprocess.DEVNULL,stdout=stdout,stderr=stderr,close_fds=True,preexec_fn=limit)
 (OUT/'handle.json').write_text(json.dumps({'pid':proc.pid,'command_sha256':EXPECTED,'started_monotonic':started}))
 try:code=proc.wait();status='completed'
 except TimeoutError:status='unknown_inspect_do_not_replay';proc.kill();code=proc.wait()
 signal.alarm(0)
(OUT/'execution.json').write_text(json.dumps({'status':status,'returncode':code,'wall_seconds':time.monotonic()-started,'local_cpu_seconds':sum(resource.getrusage(resource.RUSAGE_SELF)[:2]),'ssh_cpu_seconds':sum(resource.getrusage(resource.RUSAGE_CHILDREN)[:2])},indent=2)+'\n')
print(json.dumps({'status':status,'returncode':code,'output':str(OUT)}))
if status!='completed' or code:raise SystemExit(1)
