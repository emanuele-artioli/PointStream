def cpu():
    r = resource.getrusage(resource.RUSAGE_SELF); return r.ru_utime + r.ru_stime

def digest(path):
    h = hashlib.sha256()
    with path.open('rb') as f:
        for b in iter(lambda: f.read(1024*1024), b''): h.update(b)
    return h.hexdigest()

def datahash(data): return hashlib.sha256(data).hexdigest()

def ident(path):
    s = path.stat(); return (s.st_dev, s.st_ino, s.st_size, s.st_mtime_ns, s.st_ctime_ns)

def checkfile(path, sha, cap):
    s = path.lstat()
    if not stat.S_ISREG(s.st_mode) or s.st_size > cap: raise ValueError('invalid input ' + str(path))
    before = ident(path)
    if digest(path) != sha or ident(path) != before: raise ValueError('input mismatch ' + str(path))
    return {'bytes': s.st_size, 'sha256': sha}

def safe_name(name):
    p = PurePosixPath(name)
    return not p.is_absolute() and all(x not in ('', '.', '..') for x in p.parts)

def selected(tarpath, requests, target):
    """Each request maps exact member to a flat exclusive destination and cap."""
    seen = set()
    with tarfile.open(tarpath, 'r:') as tf:
        for m in tf:
            if m.name not in requests: continue
            name, cap = requests[m.name]
            if m.name in seen or not m.isfile() or not 0 < m.size <= cap: raise ValueError('bad selected member ' + m.name)
            src = tf.extractfile(m)
            if src is None: raise ValueError('missing member body')
            left = m.size
            with (target / name).open('xb') as out:
                while left:
                    b = src.read(min(left, 1024*1024))
                    if not b: raise ValueError('truncated member')
                    out.write(b); left -= len(b)
            seen.add(m.name)
    if seen != set(requests): raise ValueError('missing members: ' + repr(set(requests)-seen))

def snapshot(tarpath, target):
    """Extract only Python source from the hash-pinned judge source snapshot."""
    count = size = 0; seen = set()
    with tarfile.open(tarpath, 'r:*') as tf:
        for m in tf:
            if not m.name.endswith('.py'): continue
            if not safe_name(m.name) or not m.isfile() or m.name in seen: raise ValueError('unsafe source member')
            count += 1; size += m.size
            if count > 10000 or size > 150_000_000 or m.size > 4_000_000: raise ValueError('source cap')
            dest = target / m.name; dest.parent.mkdir(parents=True, exist_ok=True)
            src = tf.extractfile(m)
            if src is None: raise ValueError('missing source')
            with dest.open('xb') as out:
                left = m.size
                while left:
                    b = src.read(min(left, 1024*1024))
                    if not b: raise ValueError('short source')
                    out.write(b); left -= len(b)
            seen.add(m.name)
    for path, sha in SOURCE_HASHES.items():
        checkfile(target/path, sha, 4_000_000)

def paid_kps(path):
    b = path.read_bytes()
    if datahash(b) != PACKET_SHA or b[:5] != b'PSW8\x02': raise ValueError('packet identity')
    coded, raw = struct.unpack('>II', b[5:13])
    if coded > 4_194_304 or raw > 4_194_304: raise ValueError('packet header size')
    d = zlib.decompressobj(); text = d.decompress(b[13:13+coded], raw+1)
    if len(text) != raw or not d.eof or d.unused_data or d.unconsumed_tail: raise ValueError('packet header invalid')
    h = json.loads(text); pos = 13+coded; result = None; names = set()
    for e in h['sections']:
        name, length = e['name'], e['length']
        if name in names or type(length) is not int or length < 0 or pos+length > len(b): raise ValueError('packet section')
        names.add(name); part = b[pos:pos+length]; pos += length
        if datahash(part) != e['sha256']: raise ValueError('packet section hash')
        if name == h['profile']['kps']: result = part
    if pos != len(b) or result is None or datahash(result) != KPS_SHA: raise ValueError('packet KPS mismatch')
    return result

def inspect_data(d, count, outdir, source):

    from experiments.tennis.k6b_receiver_sam import project
    from experiments.tennis.k6b_real_base_compose import compose
    from experiments.tennis.k6b import unit_name
    cv2.setNumThreads(1)
    if np.__version__ != '1.26.4' or cv2.__version__ != '4.10.0': raise ValueError('original preparation runtime versions required')
    def array_sha(a): return datahash(np.ascontiguousarray(a).tobytes())
    kps = (d/'keypoints.kps').read_bytes()
    if datahash(kps) != KPS_SHA or kps != paid_kps(PACKET): raise ValueError('KPS differs from paid packet')
    meta, units = decode_keypoints(kps); order = meta['compositing_order']
    if meta['window'] != WID or len(units) != 2 or {u['name'] for u in units} != set(UNITS) or len(order) != 2 or {unit_name(WID, x) for x in order} != set(UNITS): raise ValueError('unit binding')
    boxes = box_maps(units); byname = {u['name']: i for i,u in enumerate(units)}
    diag = json.loads((d/'diagnostic.json').read_text())
    dr = [r for r in diag['base'] if r['window'] == 'nad000' and r['scale'] == 4]
    if len(dr)!=1 or dr[0]['ticks'] != TICKS or dr[0]['sha256'] != 'f5795d8a30277d6624b327cffae6fd0500ab8bcb196e1248b37df5a610dab0db': raise ValueError('diagnostic identity')
    judge = json.loads((d/'window.json').read_text())
    if judge['id'] != WID or judge['ticks'] != TICKS: raise ValueError('judge ticks')
    receiver = json.loads((d/'receiver.json').read_text()); runs = receiver['prediction']['runs']
    if len(runs) != 2: raise ValueError('receiver runs')
    masks = {}; crops = {}
    for unit in UNITS:
        rows = [[r for r in run['rows'] if r['name'] == unit] for run in runs]
        if any(len(row)!=1 for row in rows): raise ValueError('mask metadata binding')
        with np.load(d/(unit+'-masks.npz'), allow_pickle=False) as z:
            a = z['masks']; ticks = z['t'].tolist()
            if a.shape != (80,784,512) or a.dtype != np.bool_ or ticks != TICKS: raise ValueError('mask dimensions')
            for row in rows:
                if row[0]['ticks'] != TICKS or row[0]['output_sha256'] != array_sha(a) or row[0]['frames'] != [array_sha(x) for x in a]: raise ValueError('mask hash mismatch')
            masks[unit] = a.copy()
        with np.load(d/(unit+'@match5.npz'), allow_pickle=False) as z:
            a = z['crops']; ticks = z['t'].tolist()
            if ticks != TICKS or a.shape != (80,784,512,3) or a.dtype != np.uint8: raise ValueError('generated crops')
            crops[unit] = a.copy()
    base = np.load(d/'nad000-s4-gpu.npy', mmap_mode='r', allow_pickle=False)
    if base.shape != (80,1080,1920,3) or base.dtype != np.uint8: raise ValueError('RGB array shape')
    out = np.lib.format.open_memmap(outdir/'composite.npy', mode='w+', dtype=np.uint8, shape=(count,1080,1920,3))
    hashes = []; frame_cpu = []
    for j, t in enumerate(TICKS[:count]):
        start_cpu = cpu()
        frame = np.asarray(base[j]); reference = judge['composite_hashes'][str(t)]
        if len(reference)!=6 or array_sha(frame) != reference[2] or reference[2] != dr[0]['frame_sha256'][j]: raise ValueError('base frame '+str(t))
        player_crops = {}; player_boxes = {}; player_masks = {}
        for p in order:
            unit = unit_name(WID,p); i = byname[unit]
            player_crops[p] = crops[unit][j]; player_boxes[p] = boxes[i][t]; player_masks[p] = project(masks[unit][j], boxes[i][t])
        composed, _ = compose(frame, player_crops, player_boxes, player_masks, order)
        h = array_sha(composed)
        if h != reference[3]: raise ValueError('composite frame '+str(t))
        out[j] = composed; hashes.append(h); frame_cpu.append(cpu()-start_cpu)
    out.flush(); del out
    return {'ticks': TICKS[:count], 'composite_frame_sha256': hashes, 'frame_cpu_s': frame_cpu, 'numpy': np.__version__, 'opencv': cv2.__version__, 'python': sys.version, 'order': order}
