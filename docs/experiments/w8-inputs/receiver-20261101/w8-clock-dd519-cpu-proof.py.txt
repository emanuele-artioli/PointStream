def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def guard_tree(source):
    tree = ast.parse(source)
    fn = next(n for n in tree.body if isinstance(n, ast.FunctionDef) and n.name == 'run_isolated')
    return next(n for n in fn.body if isinstance(n, ast.For) and ast.unparse(n.target) == '(name, path)' and ast.unparse(n.iter) == 'modules.items()')


def child():
    from experiments.tennis import w8_clock_profile as profile
    from experiments.tennis import w8_cuda_system_leaves as policy
    from experiments.tennis import w8_receiver_bootstrap as bootstrap
    if os.environ.get('CUDA_VISIBLE_DEVICES') != '' or 'torch' in sys.modules:
        raise ValueError('child must have hidden GPU and no Torch')
    objects, proof = profile.prepare_child(ROOT / 'clock-identity.json', sys.argv[2],
                                            [Path(p) for p in json.loads((ROOT / 'denied.json').read_text())])
    profile.validate_receipt(proof)
    loader = json.loads((ROOT / 'loader.json').read_text())
    denied = [Path(p) for p in json.loads((ROOT / 'denied.json').read_text())]
    extra, leaves = policy.approved_leaves(denied, loader)
    output, canary = ROOT / 'child-output', ROOT / 'canary'
    paths = [output, Path(sys.prefix), *(Path(p) for p in policy.SYSTEM_GRANTS), *extra, *objects]
    ns = {n: os.readlink('/proc/self/ns/' + n) for n in ('user', 'net')}
    if sorted(name for _, name in socket.if_nameindex()) != ['lo']:
        raise ValueError('child network namespace differs')
    abi, mask = profile.restrict(policy, paths, output)
    denials = bootstrap._check_denials(denied)
    mutations = policy.mutation_probes(canary, output)
    print(json.dumps({'profile': proof, 'abi': abi, 'mask': mask, 'denials': denials,
                      'mutations': mutations, 'approved_leaves': leaves, 'namespaces': ns,
                      'cuda_visible_devices': '', 'torch_imported': False}, sort_keys=True))


def parent():

    cpus = sorted(os.sched_getaffinity(0))

    from experiments.tennis import w8_strict_receiver_worker as strict_worker

                 'animate_anyone.models.unet_3d', 'animate_anyone.pipelines.pipeline_pose2vid_long'):
        importlib.import_module(name)

    origins = strict_worker.validate_source_origins(SOURCE)
    job.check_clock_source_binding(config, origins, manifest['files'])
    bundle_path = Path('/local/users/emanuele/pointstream/cache/ab5f1b932918ae9956f6ef648e36e59c6a1fd9ef6c36a1722a1916b0da63a4c5/bundle.json')
    assert sha(bundle_path) == 'ab5f1b932918ae9956f6ef648e36e59c6a1fd9ef6c36a1722a1916b0da63a4c5'

    if ast.dump(aa_old, include_attributes=False) != ast.dump(aa_new, include_attributes=False):
        raise ValueError('original AA/k6b guard body changed')
    tree = ast.parse(inspect.getsource(receiver.run_isolated))

                 and any(isinstance(t, ast.Name) and t.id == 'clock_module' for t in n.targets))
