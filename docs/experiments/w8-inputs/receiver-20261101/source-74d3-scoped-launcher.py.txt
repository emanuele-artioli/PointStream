def alarm_handler(_signal, _frame):
    raise TimeoutError('local source publication deadline')


def limit_ssh():
    resource.setrlimit(resource.RLIMIT_CPU, (1, 1))


def main():
    signal.signal(signal.SIGALRM, alarm_handler)
    signal.alarm(130)
    old, new = OLD.read_bytes(), NEW.read_bytes()
    arg = (EVIDENCE / 'source-74d3-scoped-delta.b64.txt').read_bytes()
    program = (EVIDENCE / 'source-74d3-scoped-remote.py.txt').read_bytes()
    if (len(old) != 10741760 or hashlib.sha256(old).hexdigest() != OLD_SHA or len(new) != 4638720
            or hashlib.sha256(new).hexdigest() != NEW_SHA or hashlib.sha256(arg).hexdigest() != ARG_SHA
            or hashlib.sha256(program).hexdigest() != PROGRAM_SHA):
        raise ValueError('local frozen publication inputs changed')
    with tarfile.open(NEW, 'r:') as tar:
        entries = tar.getmembers()
        regular = [item for item in entries if item.isfile()]
        if (tar.pax_headers.get('comment') != REV or len(entries) != 249 or len(regular) != 235

                or len({item.name for item in entries}) != len(entries)):

            body = stream.read() if stream is not None else None

                raise ValueError('local source member changed: ' + name)

        raise ValueError('delta decompression bounds')
    delta = json.loads(raw)
    if (delta['schema'], delta['old_sha256'], delta['old_size'], delta['new_sha256'],

        raise ValueError('delta identities changed')
    digest = hashlib.sha256()
    position = 0
    for operation in delta['runs']:
        if operation[0] == 'r' and len(operation) == 3:
            start, length = operation[1:]
            if (type(start) is not int or type(length) is not int or start < 0 or length <= 0
                    or start % 512 or length % 512 or start + length > len(old)):
                raise ValueError('invalid delta reference')
            chunk = old[start:start + length]
        elif operation[0] == 'l' and len(operation) == 2:
            chunk = base64.b64decode(operation[1], validate=True)
            if not chunk or len(chunk) % 512:
                raise ValueError('invalid delta literal')
        else:
            raise ValueError('invalid delta opcode')

            raise ValueError('independent local delta comparison failed')
