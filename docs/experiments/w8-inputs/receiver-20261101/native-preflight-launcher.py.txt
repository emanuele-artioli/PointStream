LOCAL_ARCHIVE = Path('/private/tmp/pointstream-w8-receiver-source-5651a24b.tar')
LOCAL_OUTPUT = Path('/private/tmp/w8-native-preflight-5651a24b-20261101')

SOURCE_BASE = BASE + '/audits/w8-receiver-cpu-preflight-5651a24b-20261101'
REMOTE_OUTPUT = BASE + '/audits/w8-native-preflight-5651a24b-20261101'
REMOTE_CONFIG = BASE + '/audits/w8-native-preflight-5651a24b-20261101-config.json'
EMPTY_PYCACHE = BASE + '/audits/w8-native-preflight-5651a24b-20261101-pycache'

CONFIG_BYTES = json.dumps(CONFIG, sort_keys=True, separators=(',', ':')).encode('ascii')
CONFIG_SHA = hashlib.sha256(CONFIG_BYTES).hexdigest()

def file_sha(path):
    h = hashlib.sha256()
    with path.open('rb') as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b''):
            h.update(block)
    return h.hexdigest()


def limit_ssh():
    resource.setrlimit(resource.RLIMIT_CPU, (1, 1))


def main():
    if LOCAL_ARCHIVE.stat().st_size != 10711040 or file_sha(LOCAL_ARCHIVE) != ARCHIVE_SHA:
        raise ValueError('local physical source archive changed')
    # This Python script is passed as one safely quoted argument, not generated shell code.
    bootstrap = r'''import ctypes,hashlib,json,os,resource,signal,sys,tarfile
from pathlib import Path
resource.setrlimit(resource.RLIMIT_CPU,(1,9))
os.sched_setaffinity(0,{min(os.sched_getaffinity(0))})
source=Path(SOURCE_BASE)/'source'; archive=Path(SOURCE_BASE)/'source.tar'
local=Path(BASE); data=Path('/home/itec/emanuele/Datasets/pointstream-data')
if local.stat().st_dev==data.stat().st_dev:raise ValueError('host local filesystem required')
if os.statvfs(local).f_bavail*os.statvfs(local).f_frsize<200_000_000:raise ValueError('host local space')
if Path(REMOTE_OUTPUT).exists() or Path(REMOTE_CONFIG).exists() or Path(EMPTY_PYCACHE).exists():raise ValueError('output or config already exists')
if archive.stat().st_size!=10711040:raise ValueError('physical archive size')
h=hashlib.sha256()
with archive.open('rb') as f:
 for b in iter(lambda:f.read(1048576),b''):h.update(b)
if h.hexdigest()!=ARCHIVE_SHA:raise ValueError('physical archive SHA mismatch')
names=['experiments/__init__.py','experiments/tennis/__init__.py','experiments/tennis/w8_receiver_native_bootstrap.py','experiments/tennis/w8_receiver_native_preflight.py']
with tarfile.open(archive,'r:') as tar:
 members=tar.getmembers()
 if len(members)>5000:raise ValueError('source archive member cap')
 for name in names:
  matches=[x for x in members if x.name==name]
  if len(matches)!=1 or not matches[0].isfile() or matches[0].size>1000000:raise ValueError('invalid source member '+name)
  f=tar.extractfile(matches[0]);expected=f.read() if f else None
  if expected is None or len(expected)!=matches[0].size:raise ValueError('source member truncated')
  path=source/name
  fd=os.open(path,os.O_RDONLY|os.O_NOFOLLOW)
  try:
   stat=os.fstat(fd)
   if stat.st_size!=len(expected):raise ValueError('source file length '+name)
   observed=b''
   while len(observed)<stat.st_size:
    block=os.read(fd,min(65536,stat.st_size-len(observed)))
    if not block:raise ValueError('source file truncated')
    observed+=block
  finally:os.close(fd)
  if hashlib.sha256(observed).digest()!=hashlib.sha256(expected).digest():raise ValueError('source file SHA '+name)
config=CONFIG_BYTES
with open(REMOTE_CONFIG,'xb') as f:f.write(config)
if hashlib.sha256(config).hexdigest()!=CONFIG_SHA:raise ValueError('config SHA')
parent=os.getppid(); libc=ctypes.CDLL(None,use_errno=True)
if libc.prctl(1,signal.SIGKILL,0,0,0) or os.getppid()!=parent:raise RuntimeError('parent-death signal setup failed')
print(json.dumps({'status':'exec','pid':os.getpid(),'ppid':parent,'pgid':os.getpgrp(),'source_archive_sha256':ARCHIVE_SHA,'config_sha256':CONFIG_SHA}),flush=True)
env={'PATH':'/usr/bin:/bin','PYTHONPATH':str(source),'PYTHONNOUSERSITE':'1','PYTHONDONTWRITEBYTECODE':'1','PYTHONPYCACHEPREFIX':EMPTY_PYCACHE,'CUDA_VISIBLE_DEVICES':'','OPENBLAS_NUM_THREADS':'1','OMP_NUM_THREADS':'1','MKL_NUM_THREADS':'1','NUMEXPR_NUM_THREADS':'1','LD_LIBRARY_PATH':TREE+'/lib','TMPDIR':BASE,'TMP':BASE,'TEMP':BASE,'XDG_CACHE_HOME':BASE}
python=TREE+'/bin/python'
os.execve(python,[python,'-B','-m','experiments.tennis.w8_receiver_native_preflight','--config',REMOTE_CONFIG,'--config-sha256',CONFIG_SHA],env)
'''
    values = {'SOURCE_BASE': SOURCE_BASE, 'BASE': BASE, 'REMOTE_OUTPUT': REMOTE_OUTPUT,
              'REMOTE_CONFIG': REMOTE_CONFIG, 'EMPTY_PYCACHE': EMPTY_PYCACHE, 'ARCHIVE_SHA': ARCHIVE_SHA,
              'CONFIG_BYTES': CONFIG_BYTES, 'CONFIG_SHA': CONFIG_SHA, 'TREE': TREE}
    program = '\n'.join(f'{key}={value!r}' for key, value in values.items()) + '\n' + bootstrap
    command = ' '.join(shlex.quote(x) for x in ['/bin/sh', '-c', 'ulimit -S -t 1; exec "$@"', '_',
                                                 '/usr/bin/timeout', '--signal=TERM', '--kill-after=2',
                                                 '140s', '/usr/bin/python3', '-B', '-c', program])
    LOCAL_OUTPUT.mkdir(mode=0o700, exist_ok=False)
    (LOCAL_OUTPUT/'config.json').write_bytes(CONFIG_BYTES)
    (LOCAL_OUTPUT/'remote-program.py.txt').write_text(program)
    started = time.monotonic()
    with (LOCAL_OUTPUT/'stdout.log').open('xb') as stdout, (LOCAL_OUTPUT/'stderr.log').open('xb') as stderr:
        proc = subprocess.Popen(['ssh', '-o', 'BatchMode=yes', '-o', 'ConnectTimeout=5', 'gpu6', command],
                                stdin=subprocess.DEVNULL, stdout=stdout, stderr=stderr, close_fds=True,
                                start_new_session=True, preexec_fn=limit_ssh)
        handle = {'pid': proc.pid, 'pgid': proc.pid, 'started_monotonic': started,
                  'remote_deadline_seconds': 140, 'local_deadline_seconds': 150,
                  'remote_output': REMOTE_OUTPUT, 'source_sha256': ARCHIVE_SHA, 'config_sha256': CONFIG_SHA}
        (LOCAL_OUTPUT/'handle.json').write_text(json.dumps(handle, indent=2))
        status = 'completed'
        try:
            code = proc.wait(timeout=150)
        except subprocess.TimeoutExpired:
            status = 'unknown_inspect_do_not_replay'
            try:
                os.killpg(proc.pid, signal.SIGKILL)
            except ProcessLookupError:
                pass
            code = proc.wait(timeout=5)
        result = {**handle, 'status': status, 'returncode': code, 'wall_seconds': time.monotonic()-started,
                  'local_cpu_seconds': sum(resource.getrusage(resource.RUSAGE_SELF)[:2]),
                  'ssh_cpu_seconds': sum(resource.getrusage(resource.RUSAGE_CHILDREN)[:2]),
                  'cpu_reservation_s': 20, 'revision': REVISION}
        (LOCAL_OUTPUT/'execution.json').write_text(json.dumps(result, indent=2))
        print(json.dumps({'status': status, 'returncode': code, 'local_output': str(LOCAL_OUTPUT)}))
        if status != 'completed' or code:
            raise SystemExit(1)


if __name__ == '__main__':
    main()
