"""Replay archived arithmetic only. Run from the PointStream repository root."""
import ast
import hashlib
import json
import statistics
import subprocess
import sys
from pathlib import Path

import numpy
import scipy
from experiments.tennis import k5b1b

root = Path(__file__).resolve().parent
receipt = json.loads((root / "arithmetic-audit.json").read_text())
for name, identity in receipt["selected_input_files"].items():
    data = (root / name).read_bytes()
    assert len(data) == identity["bytes"]
    assert hashlib.sha256(data).hexdigest() == identity["sha256"]
for qualified in receipt["ast_equivalent_historical_functions"]:
    filename, name = qualified.split(":")
    old = ast.parse(subprocess.check_output(["git", "show", receipt["source_revision"] + ":" + filename], text=True))
    current = ast.parse(Path(filename).read_text())
    def function(tree):
        return next(n for n in tree.body if isinstance(n, ast.FunctionDef) and n.name == name)
    assert ast.dump(function(old), include_attributes=False) == ast.dump(function(current), include_attributes=False)
rows = [json.loads(p.read_text()) for p in sorted(root.glob("tracknet__*.json"))]
record = json.loads((root / "k5b1b.json").read_text())
assert len(rows) == 4
checks = k5b1b.validate_rows(rows, record)
assert checks == receipt["original_validator_checks"] and all(checks.values())
for variant, expected in receipt["recomputed_qp0_equal_window_means"].items():
    values = [p["variants"][variant] for r in rows for p in r["points"] if p["codec"] == "dcvc-uf-htl" and p["param"] == 0]
    assert len(values) == expected["windows"]
    assert statistics.mean(v["kbps"] for v in values) == expected["mean_total_kbps"]
    assert statistics.mean(v["detection"]["detection_rate"] for v in values) == expected["mean_balltrack_recall"]
for row in rows:
    result = k5b1b.window_report(row)
    key = "streak@balltrack-dcvc-uf-htl"
    actual = {"dists_bd_rate_pooled": result["dists"][key]["pooled"],
              "frame_psnr_bd_rate_dcvc": result["frame_psnr"][key]["dcvc-uf-htl"],
              "historical_detection_verdict": result["detection"][key],
              "historical_gate_checks": result["passes"][key]}
    assert actual == receipt["recomputed_streak_dcvc"][row["id"]]
print(json.dumps({"passed": True, "windows": len(rows), "archived_checks": len(checks),
                  "python": sys.version.split()[0], "numpy": numpy.__version__, "scipy": scipy.__version__,
                  "scope": receipt["scope"]}))
